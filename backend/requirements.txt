# Runtime dependencies for the OptionQ API container.
#
# Every version here is the version the test suite actually passes against in
# backend/.venv. The previous pins were a full major version behind on the
# numerical stack (pandas 2.2.3 vs 3.0.2, numpy 1.26.4 vs 2.4.4, yfinance
# 0.2.44 vs 1.2.2, arch 7.1.0 vs 8.0.0, scikit-learn 1.5.2 vs 1.8.0), so the
# deployed container ran different libraries from the ones any test had
# exercised — and yfinance 0.2.x has neither the chain API nor the
# curl_cffi-backed session the market-data layer now depends on.
#
# Nothing is listed here that `backend/` does not import. Dev/test extras live
# in backend/requirements-dev.txt, which includes this file.

# Web framework
fastapi==0.135.3
uvicorn[standard]==0.44.0
pydantic==2.13.1
pydantic-settings==2.13.1

# Market data.
# curl_cffi is pinned explicitly although nothing in backend/ imports it:
# yfinance's own transport is built on it and raises YFDataException when
# handed a plain requests.Session, so an incompatible curl_cffi silently costs
# us every quote yfinance fetches. (The browser-headers session in
# backend/data/market_data.py is a plain requests.Session and is used only for
# the direct Yahoo chart API, which yfinance does not mediate — an earlier
# version of this comment had those two the wrong way round.)
yfinance==1.2.2
curl_cffi==0.15.0
requests==2.33.1

# Numerics
pandas==3.0.2
numpy==2.4.4
scipy==1.17.1

# ML — GMM regime detection (scikit-learn) and GARCH(1,1) forecasting (arch).
# joblib is imported directly by backend/ml/regime/gmm_detector.py to persist
# the fitted model, so it is a direct dependency, not just a sklearn transitive.
scikit-learn==1.8.0
joblib==1.5.3
arch==8.0.0

# LLM explainer. Optional at runtime: without a key the pipeline returns
# deterministic explanations and says the LLM was unavailable.
anthropic==0.95.0
huggingface_hub==1.17.0

# Vertex AI authentication for the Gemini provider.
#
# Vertex has no API key: every call carries an OAuth bearer token, so the one
# thing we cannot do without a library is mint one. google-auth is that
# library and nothing more — it resolves Application Default Credentials (a
# developer's gcloud login locally, the attached service account in Cloud Run)
# and signs the token. The REST call itself stays on `requests`.
#
# The heavier google-genai / google-cloud-aiplatform SDKs were rejected: both
# pull grpcio and protobuf, which have no cp313 wheels for every platform we
# build and would have broken the property that the whole closure installs
# from wheels with zero source builds. google-auth's own closure (pyasn1,
# pyasn1-modules, cryptography) is wheel-only and ~5MB; grpcio and protobuf
# appear only under its `grpc`/`testing`/`aiohttp` extras, which we do not
# install.
google-auth==2.62.0

# Scheduler — fast(24h) + main(monthly) model retrains
apscheduler==3.11.2

# Config
python-dotenv==1.2.2

# ── Transitive pins ──────────────────────────────────────────────────────────
# wrapt is not imported by backend/; it arrives via arch → statsmodels →
# formulaic. Pinned because formulaic declares `wrapt>=1.17.0rc1` for Python
# >= 3.13, and a specifier that names a prerelease makes pip willing to resolve
# prereleases for that requirement — so the closure picked up wrapt 2.6.0.dev1,
# a development build, in preference to the 2.5.2 stable release. A dev
# prerelease has no release guarantees and can change behaviour between
# identical-looking builds; it does not belong in the dependency graph of
# something that prices hedges. 2.5.2 satisfies formulaic's constraint.
wrapt==2.5.2
